## 用法

<p>
  DatePicker 支持三种选择模式：`single`、`multiple` 和 `range`。
</p>

```tsx
import { useState } from "react";
import { DatePicker } from "@cloudflare/kumo";

export default function Example() {
  const [date, setDate] = useState<Date>();

  return <DatePicker mode="single" selected={date} onChange={setDate} />;
}
```

## 完整的 Popover 示例

<p>以下是一个完整示例，展示如何将 DatePicker 与 Popover 组合使用：</p>

```tsx
import { useState } from "react";
import { DatePicker, Popover, Button } from "@cloudflare/kumo";
import { CalendarDotsIcon } from "@phosphor-icons/react";

export function DatePickerDropdown() {
  const [date, setDate] = useState<Date>();

  return (
    <Popover>
      <Popover.Trigger
        render={<Button variant="outline" icon={CalendarDotsIcon} />}
      >
        {date ? date.toLocaleDateString() : "Pick a date"}
      </Popover.Trigger>
      <Popover.Content className="p-3">
        <DatePicker
          mode="single"
          selected={date}
          onChange={(d) => setDate(d)}
        />
      </Popover.Content>
    </Popover>
  );
}
```

## API 参考

<p>
  DatePicker 将所有属性转发给
  <a
    href="https://daypicker.dev/api/interfaces/PropsBase"
    target="_blank"
    rel="noopener noreferrer"
  >
    react-day-picker
  </a>
  。关键属性包括：
</p>
<ul class="mt-4 list-disc list-inside space-y-2">
  <li>
    `mode` — `"single" | "multiple" | "range"` — 选择模式（必填）
  </li>
  <li>`selected` — 当前选中的日期（们）</li>
  <li>`onChange` — 选择变化时的回调</li>
  <li>`numberOfMonths` — 显示的月份数量</li>
  <li>`disabled` — 不可选择的日期</li>
  <li>`min` / `max` — 最小/最大选择约束</li>
  <li>`footer` — 渲染在日历下方的内容</li>
  <li>`locale` — 用于国际化的 date-fns locale</li>
  <li>`className` — 额外的 CSS 类</li>
</ul>
<p class="mt-4 text-kumo-subtle text-sm">
  完整 API 参见
  <a
    href="https://daypicker.dev/docs"
    target="_blank"
    rel="noopener noreferrer"
  >
    react-day-picker 文档
  </a>
  。
</p>

### 与 react-day-picker 的差异

  <p>
    为了与其他 Kumo 表单组件保持一致，DatePicker 使用 `onChange` 而非 react-day-picker 的 `onSelect`。
    类型推断保持完整——TypeScript 会根据 `mode` 属性正确窄化回调签名。
  </p>
